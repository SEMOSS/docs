# ExampleReactor

**Description:** 
``` markdown
This is an example reactor with required and optional parameters
```

## Required Keys
- **query**
- **database**

## Optional Keys
- **limit**
- **format**

## Usage
```javascript
ExampleReactor(query=[<query>], database=[<database>], limit=[<limit>], format=[<format>])
```

## Try It Out

<div className="try-it-out-section">

You can test this reactor directly using the `runPixelSync` API endpoint. Here are examples in different formats:

### cURL
```bash
curl -X POST "https://workshop.semoss.com/semoss-ai-dev/Monolith/api/engine/runPixel" \
  -H "Content-Type: application/x-www-form-urlencoded" \
  -H "Accept: application/json" \
  -H "Authorization: Basic ****" \
  -d "expression=ExampleReactor(query%3D%5B%3Cquery%3E%5D%2C%20database%3D%5B%3Cdatabase%3E%5D%2C%20limit%3D%5B%3Climit%3E%5D%2C%20format%3D%5B%3Cformat%3E%5D)"
```

### JavaScript (Fetch API)
```javascript
const response = await fetch('https://workshop.semoss.com/ semoss-ai-dev/Monolith/api/engine/runPixel', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/x-www-form-urlencoded',
    'Accept': 'application/json',
    'Authorization': 'Basic ****'
  },
  body: new URLSearchParams({
    expression: 'ExampleReactor(query=[<query>], database=[<database>], limit=[<limit>], format=[<format>])'
  })
});

const result = await response.json();
console.log(result.pixelReturn[0].output);
```

### Python (requests)
```python
import requests

url = "https://workshop.semoss.com/semoss-ai-dev/Monolith/api/engine/runPixel"
headers = {
    "Content-Type": "application/x-www-form-urlencoded",
    "Accept": "application/json",
    "Authorization": "Basic ****"
}
data = {
    "expression": "ExampleReactor(query=[<query>], database=[<database>], limit=[<limit>], format=[<format>])"
}

response = requests.post(url, headers=headers, data=data)
result = response.json()
print(result["pixelReturn"][0]["output"])
```

  function testReactorExampleReactor() {
    const params = [];
    
    const queryValue = document.getElementById('query').value;
    if (queryValue && queryValue.trim()) {
      params.push('query=[' + queryValue.trim() + ']');
    }
    
    const databaseValue = document.getElementById('database').value;
    if (databaseValue && databaseValue.trim()) {
      params.push('database=[' + databaseValue.trim() + ']');
    }
    
    const limitValue = document.getElementById('limit').value;
    if (limitValue && limitValue.trim()) {
      params.push('limit=[' + limitValue.trim() + ']');
    }
    
    const formatValue = document.getElementById('format').value;
    if (formatValue && formatValue.trim()) {
      params.push('format=[' + formatValue.trim() + ']');
    }
    
    const expression = params.length > 0 
      ? 'ExampleReactor(' + params.join(', ') + ')'
      : 'ExampleReactor()';
    
    const outputElement = document.getElementById('response-output-examplereactor');
    
    const requestBody = 'expression=' + encodeURIComponent(expression);
    
    const requestDetails = `Endpoint: POST https://workshop.semoss.com/semoss-ai-dev/Monolith/api/engine/runPixel

Body:
${requestBody}

Generated Pixel Expression:
${expression}

cURL Command:
curl -X POST "https://workshop.semoss.com/semoss-ai-dev/Monolith/api/engine/runPixel" \\
